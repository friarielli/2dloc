% Theorem environments, all numbered in one sequence. thmtools gives each
% environment its own (aliased) counter and sets the cleveref names from
% refname (cleveref derives \Cref from it), so \cref works without cleveref guessing the type.

\declaretheorem[style=plain, name=Proposition, sibling=thm, refname={Proposition,Propositions}]{prop}
\declaretheorem[style=plain, name=Lemma, sibling=thm, refname={Lemma,Lemmas}]{lem}
\declaretheorem[style=plain, name=Corollary, sibling=thm, refname={Corollary,Corollaries}]{cor}
\declaretheorem[style=plain, name=Conjecture, sibling=thm, refname={Conjecture,Conjectures}]{conj}
\declaretheorem[style=plain, name=Example, sibling=thm, refname={Example,Examples}]{exmp}
\declaretheorem[style=definition, name=Definition, sibling=thm, refname={Definition,Definitions}]{defi}
\declaretheorem[style=definition, name=Construction, sibling=thm, refname={Construction,Constructions}]{constru}
\declaretheorem[style=remark, name=Remark, sibling=thm, refname={Remark,Remarks}]{rmk}
