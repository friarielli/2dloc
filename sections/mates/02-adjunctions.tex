\subsection{Morphisms between adjunctions}
\label{subsec:mates-morphisms}

\gutentag{0015}%
The objects of $\func(\Adj,\E)$ are the adjunctions in $\E$
(\cref{def:adjunction}), and its $1$-morphisms are natural transformations of
adjunctions. We first describe $\func(\Adj,\E)$ inside $\func(\Delta^{1},\E)$:
this is the mate lemma. We then show that, when $\E$ has lax pullbacks,
adjunctions can equivalently be defined as equivalences of lax pullbacks,
without reference to $\Adj$, and that morphisms between adjunctions so defined
are pairs of mate squares, neither of which needs to be invertible.

\subsubsection*{The mate lemma}

\begin{lem}[mates]
\label{lem:mates}
\gutentag{0016}%
Let $\E$ be an $(\infty,2)$-category.
\begin{enumerate}
\item
$\ell^{*}\colon \func(\Adj,\E) \to \func(\Delta^{1},\E)$
is a locally full inclusion. It exhibits $\func(\Adj,\E)$ as the locally full
sub-$(\infty,2)$-category whose objects are the left adjoints $u\colon x \to y$,
and whose $1$-morphisms from $u$ to $u'$ are the commutative squares for which
the lax square
\begin{equation*}
\begin{tikzcd}
x \ar[r, "f_{0}"] \ar[d, "u"'] & x' \ar[d, "u'"] \ar[dl, Rightarrow,
shorten=3mm, "\varphi"'] \\
y \ar[r, "f_{1}"'] & y'
\end{tikzcd}
\qquad \varphi\colon u'f_{0} \simeq f_{1}u \ ,
\end{equation*}
is right Beck--Chevalley:
$\rmate{\varphi}\colon f_{0}u^{R} \Rightarrow u'^{R}f_{1}$
is invertible.
\item Dually,
$\rho^{*}\colon \func(\Adj,\E) \to \func(\Delta^{1},\E)$
is a locally full inclusion. It exhibits $\func(\Adj,\E)$ as the locally full
sub-$(\infty,2)$-category whose objects are the right adjoints $u\colon x \to
y$, and whose $1$-morphisms from $u$ to $u'$ are the commutative squares for
which the lax square
\begin{equation*}
\begin{tikzcd}
x \ar[r, "u"] \ar[d, "f_{0}"'] & y \ar[d, "f_{1}"] \ar[dl, Rightarrow,
shorten=3mm, "\psi"'] \\
x' \ar[r, "u'"'] & y'
\end{tikzcd}
\qquad \psi\colon f_{1}u \simeq u'f_{0} \ ,
\end{equation*}
is left Beck--Chevalley:
$\lmate{\psi}\colon u'^{L}f_{1} \Rightarrow f_{0}u^{L}$
is invertible.
\end{enumerate}
\end{lem}

\begin{proof}
\gutentag{0017}%
We prove (1); (2) is (1) for $\E^{\mathrm{co}}$, since
$\Adj^{\mathrm{co}} \simeq \Adj$
exchanges $\ell$ and $\rho$, while passing to $\E^{\mathrm{co}}$ exchanges left
and right adjoints and turns right mates into left mates of the transposed
squares.

Let
$\func(\Adj,\E)^{\mathrm{colax}}$
be the $(\infty,2)$-category of adjunctions in $\E$ and colax morphisms of
adjunctions \cite[Def.~4.2]{haugseng2021laxmonads},
i.e.\ of colax transformations
$\Adj \otimes \Delta^{1} \to \E$
for the Gray tensor product.
By \cite[Cor.~2.8.12]{abellangagnahaugseng2024straightening},
$\func(K,\E)$ is the wide locally full sub-$(\infty,2)$-category of
$\func(K,\E)^{\mathrm{colax}}$
on the strong transformations, those whose naturality $2$-morphisms are
invertible, naturally in $K$; so $\ell^{*}$ on $\func(\Adj,\E)$ is the
restriction of
$\ell^{*}\colon \func(\Adj,\E)^{\mathrm{colax}} \to
\func(\Delta^{1},\E)^{\mathrm{colax}}$
to strong transformations.
By \cite[Cor.~4.8]{haugseng2021laxmonads}, the latter is an equivalence from
$\func(\Adj,\E)^{\mathrm{colax}}$
onto the full sub-$(\infty,2)$-category of $\func(\Delta^{1},\E)$ on the left
adjoints. Its proof uses \cite[Thm.~4.6]{haugseng2021laxmonads},
the characterization of adjoints in $(\infty,2)$-categories of lax
transformations, whose proof is completed in \cite[Cor.~5.2.10,
Rem.~5.2.11]{abellangagnahaugseng2024straightening}.
So
$\ell^{*}\colon \func(\Adj,\E) \to \func(\Delta^{1},\E)$
is the composite of a locally full inclusion,
an equivalence and a full inclusion, hence a locally full inclusion,
and its objects are the left adjoints.

A $1$-morphism of $\func(\Delta^{1},\E)$ between left adjoints $u$,
$u'$ is a commutative square $\varphi$ as in (1).
Under the equivalence of \cite[Cor.~4.8]{haugseng2021laxmonads} it corresponds
to the colax morphism of adjunctions whose naturality square at $l$ is $\varphi$
and whose naturality square at $r$ is $\rmate{\varphi}$
\cite[Cor.~4.9]{haugseng2021laxmonads}. As $\Adj$ is generated by $l$ and $r$
under composition, and the naturality $2$-morphism of a colax transformation at
a composite is the pasting of those at the factors,
this colax morphism is strong if and only if $\rmate{\varphi}$ is invertible.
\end{proof}

\begin{rmk}
\label{rmk:mates-2cat}
\gutentag{0018}%
For $2$-categories, \cref{lem:mates} is the classical fact that a morphism of
adjunctions is determined by its left adjoint part,
the right adjoint part being the mate \cite{kellystreet1974review}.
For $(\infty,2)$-categories, we know of no reference stating \cref{lem:mates}
itself; the proof above deduces it from the lax version
\cite[Cor.~4.8]{haugseng2021laxmonads}, in which $\ell^{*}$, resp.\ $\rho^{*}$,
is an equivalence from adjunctions with colax, resp.\ lax,
morphisms onto the full sub-$(\infty,2)$-category of $\func(\Delta^{1},\E)$ on
the left, resp.\ right, adjoints. We will give a proof from the Riehl--Verity
theorem and \cite[Cor.~5.2.12]{abellangagnahaugseng2024straightening} in
\cref{rmk:mates-alternative}.
\end{rmk}

\subsubsection*{Adjunctions via lax pullbacks}

\gutentag{0019}%
\textbf{Notation.} For $1$-morphisms $f\colon X \to Y$ and $g\colon Z \to Y$ of
$\E$ we write
$(f \downarrow g) := X \ratimes_{Y} Z$
for their lax pullback, with projections $p$, $q$ and $2$-morphism
$\alpha\colon fp \Rightarrow gq$,
and
$(u,v,\varphi)\colon T \to (f \downarrow g)$
for the $1$-morphism given by $u\colon T \to X$, $v\colon T \to Z$ and
$\varphi\colon fu \Rightarrow gv$.
For $\E = \Cat$, $(f \downarrow g)$ is the comma $\infty$-category.
For $f\colon X \to Y$ and $g\colon Y \to X$ we write
$(f \downarrow Y) := (f \downarrow \mathrm{id}_{Y})$
and
$(X \downarrow g) := (\mathrm{id}_{X} \downarrow g)$;
both are spans from $X$ to $Y$. They receive the $1$-morphisms
\begin{equation*}
i := (\mathrm{id}_{X},f,\mathrm{id}_{f})\colon X \to (f \downarrow Y) \ ,
\qquad j := (g,\mathrm{id}_{Y},\mathrm{id}_{g})\colon Y \to (X \downarrow g) \ .
\end{equation*}

\gutentag{001A}%
Let
$\Lambda := \{0 \leftarrow 2 \rightarrow 1\}$
be the walking span and
$\Span(\E) := \func(\Lambda,\E)$
the $(\infty,2)$-category of spans in $\E$. We write a span as
$S = (X \xleftarrow{p} E \xrightarrow{q} Y)$,
a span from $X$ to $Y$, and $\Span_{\E}(X,Y)$ for the fibre of
$\Span(\E) \to \E \times \E$,
evaluation at $0$ and $1$, over $(X,Y)$.
A $1$-morphism $S \to S'$ of $\Span(\E)$ over
$(a,b) \in \Hom(X,X') \times \Hom(Y,Y')$
is a \emph{map of spans}: a $1$-morphism $c\colon E \to E'$ with equivalences
$p'c \simeq ap$ and $q'c \simeq bq$. By
\cite[Prop.~4.1.2]{abellangagnahaugseng2024straightening},
applied to strong transformations,
\begin{equation*}
\Hom_{\Span(\E)}(S,S') \simeq \Hom(X,X') \times_{\Hom(E,X')} \Hom(E,E')
\times_{\Hom(E,Y')} \Hom(Y,Y') \ ,
\end{equation*}
and the hom-$\infty$-categories of $\Span_{\E}(X,Y)$ are the fibres over
$(\mathrm{id}_{X},\mathrm{id}_{Y})$.
The underlying $\infty$-category of $\Span_{\E}(X,Y)$ is the fibre of
$\func(\Lambda,\iota_{1}\E)$
over $(X,Y)$, so a $1$-morphism of $\Span_{\E}(X,Y)$ is an equivalence if and
only if it is one in $\E$.

\begin{lem}
\label{lem:comma-fibrewise}
\gutentag{001B}%
Let $F\colon A \to C$, $G\colon B \to C$,
$F'\colon A \to C'$ and $G'\colon B \to C'$ be functors of $\infty$-categories.
A functor
$(F \downarrow G) \to (F' \downarrow G')$
over $A \times B$ is an equivalence if and only if it induces equivalences
$\mapp(Fa,Gb) \to \mapp(F'a,G'b)$
on the fibres over all $(a,b)$.
\end{lem}

\begin{proof}
\gutentag{001C}%
$(F \downarrow G) \to A \times B$
is the pullback of
$\func(\Delta^{1},C) \to C \times C$
along $F \times G$, hence a bifibration \cite[Cor.~2.4.7.11]{htt},
with fibre $\mapp(Fa,Gb)$ over $(a,b)$. A functor over $A \times B$ between
bifibrations that is an equivalence on fibres is an equivalence
\cite[Prop.~2.4.7.6]{htt}.
\end{proof}

\begin{defi}
\label{def:transposing-adjunction}
\gutentag{001D}%
Let $\E$ have lax pullbacks.
\begin{enumerate}
\item A \emph{transposing adjunction} between $f\colon X \to Y$ and $g\colon Y
\to X$ is an equivalence
$e\colon (f \downarrow Y) \xrightarrow{\simeq} (X \downarrow g)$
in $\Span_{\E}(X,Y)$.
\item Let $\TAdj(\E)$ be the full sub-$(\infty,2)$-category of
$\func(\Delta^{1},\Span(\E))$
on the transposing adjunctions. A $1$-morphism $(c,d)\colon e \to e'$ of
$\TAdj(\E)$, from a transposing adjunction between $f$ and $g$ to one between
$f'\colon X' \to Y'$ and $g'\colon Y' \to X'$,
consists of maps of spans $c$ and $d$ over some
$(a,b) \in \Hom(X,X') \times \Hom(Y,Y')$
and an equivalence $de \simeq e'c$:
\begin{equation*}
\begin{tikzcd}
(f \downarrow Y) \ar[r, "e"] \ar[d, "c"'] & (X \downarrow g) \ar[d, "d"] \\
(f' \downarrow Y') \ar[r, "e'"'] & (X' \downarrow g')
\end{tikzcd}
\end{equation*}
By \cite[Prop.~4.1.2]{abellangagnahaugseng2024straightening} again,
its hom-$\infty$-category is the pullback
\begin{equation*}
\Hom_{\TAdj(\E)}(e,e') \simeq \Hom_{\Span(\E)}\bigl((f \downarrow Y),(f'
\downarrow Y')\bigr) \times_{\Hom_{\Span(\E)}((f \downarrow Y),(X' \downarrow
g'))} \Hom_{\Span(\E)}\bigl((X \downarrow g),(X' \downarrow g')\bigr)
\end{equation*}
along $e'_{\natural}$ and $e^{\natural}$, over
$\Hom(X,X') \times \Hom(Y,Y')$.
\end{enumerate}
\end{defi}

\gutentag{001E}%
For $\E = \Cat$, $(f \downarrow Y)$ and $(X \downarrow g)$ are bifibrations over
$X \times Y$ with fibres $\mapp_{Y}(fx,y)$ and $\mapp_{X}(x,gy)$,
so by \cref{lem:comma-fibrewise} a transposing adjunction is a functor over $X
\times Y$ inducing equivalences
$\mapp_{Y}(fx,y) \simeq \mapp_{X}(x,gy)$.
The name is taken from Riehl--Shulman's synthetic setting
\cite[Def.~11.1]{riehlshulman2017synthetic}.

\begin{thm}[adjunctions via lax pullbacks]
\label{thm:comma-adjunctions}
\gutentag{001F}%
Let $\E$ be an $(\infty,2)$-category with lax pullbacks.
\begin{enumerate}
\item \emph{Adjunctions are transposing adjunctions.} Let $f\colon X \to Y$ and
$g\colon Y \to X$. Every $1$-morphism
$(f \downarrow Y) \to (X \downarrow g)$
of $\Span_{\E}(X,Y)$ is, for a unique
$\eta\colon \mathrm{id}_{X} \Rightarrow gf$,
the map
\begin{equation*}
e_{\eta}\colon (f \downarrow Y) \to (X \downarrow g) \ ,
\qquad (u,v,\varphi\colon fu \Rightarrow v) \mapsto (u,v,g\varphi \circ \eta
u\colon u \Rightarrow gv) \ ;
\end{equation*}
that is,
$\eta \mapsto e_{\eta}$
is an equivalence
$\mapp(\mathrm{id}_{X},gf) \simeq \Hom_{\Span_{\E}(X,Y)}((f \downarrow Y),(X
\downarrow g))$.
The map $e_{\eta}$ is a transposing adjunction if and only if $\eta$ is the unit
of an adjunction $f \dashv g$, and
$A \mapsto e_{A(\eta)}$
is an equivalence from the space of adjunctions $A$ with $A(l) = f$ and $A(r) =
g$ to the space of transposing adjunctions between $f$ and $g$.
In particular, $f$ is a left adjoint if and only if there is a transposing
adjunction between $f$ and some $g$, and $g$ is then a right adjoint of $f$.
\item \emph{Morphisms are pairs of mate squares.} Let $e$ and $e'$ be
transposing adjunctions between $f$, $g$ and between $f'$, $g'$,
and let $f \dashv g$ and $f' \dashv g'$ be the corresponding adjunctions.
Every $1$-morphism $(c,d)\colon e \to e'$ of $\TAdj(\E)$ over $(a,b)$ has two
lax squares
\begin{equation*}
\begin{tikzcd}
X \ar[r, "a"] \ar[d, "f"'] & X' \ar[d, "f'"] \ar[dl, Rightarrow, shorten=3mm,
"\sigma"'] \\
Y \ar[r, "b"'] & Y'
\end{tikzcd}
\qquad\qquad
\begin{tikzcd}
Y \ar[r, "g"] \ar[d, "b"'] & X \ar[d, "a"] \ar[dl, Rightarrow, shorten=3mm,
"\tau"'] \\
Y' \ar[r, "g'"'] & X'
\end{tikzcd}
\end{equation*}
given by
$ci \simeq (a,bf,\sigma)$
and
$dj \simeq (ag,b,\tau)$.
The lax square $\tau$ is the right mate of $\sigma$,
and $(c,d) \mapsto \sigma$ and $(c,d) \mapsto \tau$ are equivalences of
$\infty$-categories over
$\Hom(X,X') \times \Hom(Y,Y')$
\begin{equation*}
(f'_{\natural} \downarrow f^{\natural}) \xleftarrow{\ \simeq\ }
\Hom_{\TAdj(\E)}(e,e') \xrightarrow{\ \simeq\ } (g^{\natural} \downarrow
g'_{\natural})
\end{equation*}
onto the $\infty$-categories of lax squares $\sigma$, resp.\ $\tau$.
\end{enumerate}
\end{thm}

\gutentag{0020}%
So when $\E$ has lax pullbacks, an adjunction can be defined as a transposing
adjunction, without reference to $\Adj$. For $\E = \Cat$, $e_{\eta}$ is
$\varphi \mapsto g\varphi \circ \eta_{x}$
on the fibres, so (1) says that $\eta$ is the unit of an adjunction if and only
if it is a unit transformation in the sense of \cite[Def.~5.2.2.7]{htt}.
Morphisms of adjunctions can then be defined as the $1$-morphisms of
$\TAdj(\E)$, which also provides their composition and $2$-morphisms.
By \cref{thm:comma-adjunctions}(2), such a morphism is a pair of lax squares
$(\sigma,\rmate{\sigma})$,
neither of which needs to be invertible, and it is determined by either of them.
The $1$-morphisms of $\func(\Adj,\E)$ correspond to the pairs in which both are
invertible (\cref{lem:mates}), and the (co)lax morphisms of adjunctions of
\cite[Def.~4.2]{haugseng2021laxmonads} to those in which one of them is
invertible \cite[Cor.~4.8, 4.9]{haugseng2021laxmonads}.

\begin{proof}[Proof of \cref{thm:comma-adjunctions}]
\gutentag{0021}%
We write $p$, $q$,
$\alpha\colon fp \Rightarrow q$
for the structure of $(f \downarrow Y)$ and $\bar p$, $\bar q$,
$\beta\colon \bar p \Rightarrow g\bar q$
for that of $(X \downarrow g)$. By the universal property of lax pullbacks,
$\Hom(T,(h \downarrow k))$
is the comma $\infty$-category
$(h_{\natural} \downarrow k_{\natural})$,
naturally in $T$.

\emph{Step 1: restriction along $i$ and $j$.} We have $pi = \mathrm{id}_{X}$,
$qi = f$ and
$\alpha i = \mathrm{id}_{f}$.
The pair
$(\mathrm{id}_{p},\alpha)$
is a $2$-morphism
$ip = (p,fp,\mathrm{id}_{fp}) \Rightarrow (p,q,\alpha) = \mathrm{id}$,
and it is the counit of an adjunction $i \dashv p$ with identity unit
\cite[Thm.~4.3.11]{riehlverity2016adjunctions}, as the triangle identities
$p(\mathrm{id}_{p},\alpha) = \mathrm{id}_{p}$
and
$(\mathrm{id}_{p},\alpha)i = \mathrm{id}_{i}$
hold. By \eqref{eq:hom-adjunctions},
$p^{\natural} \dashv i^{\natural}$
with identity unit, so restriction along $i$ is an equivalence
\begin{equation}
\label{eq:comma-yoneda}
\mapp(sp,t) \xrightarrow{\ \simeq\ } \mapp(s,ti) \ ,
\qquad \theta \mapsto \theta i \ ,
\end{equation}
for all $s\colon X \to T$ and
$t\colon (f \downarrow Y) \to T$
\cite[Prop.~5.2.2.8]{htt}. Dually, $\bar p j = g$,
$\bar q j = \mathrm{id}_{Y}$
and
$\beta j = \mathrm{id}_{g}$,
and
$(\beta,\mathrm{id}_{\bar q})\colon \mathrm{id} \Rightarrow j\bar q$
is the unit of an adjunction $\bar q \dashv j$ with identity counit,
so restriction along $j$ is an equivalence
$\mapp(t,s\bar q) \simeq \mapp(tj,s)$
for all $s\colon Y \to T$ and
$t\colon (X \downarrow g) \to T$.

\emph{Step 2: maps between lax pullbacks.} Let $(h \downarrow k)$ be $(f'
\downarrow Y')$ or $(X' \downarrow g')$,
with $h\colon X' \to Z'$ and $k\colon Y' \to Z'$. Restriction along $i$,
resp.\ $j$, gives equivalences over
$\Hom(X,X') \times \Hom(Y,Y')$
\begin{equation}
\label{eq:maps-of-commas}
\Hom_{\Span(\E)}\bigl((f \downarrow Y),(h \downarrow k)\bigr) \simeq
(h_{\natural} \downarrow k_{\natural}f^{\natural}) \ ,
\qquad \Hom_{\Span(\E)}\bigl((X \downarrow g),(h \downarrow k)\bigr) \simeq
(h_{\natural}g^{\natural} \downarrow k_{\natural}) \ .
\end{equation}
Indeed,
$\Hom_{\Span(\E)}((f \downarrow Y),(h \downarrow k))$
is the pullback of
$\Hom((f \downarrow Y),(h \downarrow k)) \simeq (h_{\natural} \downarrow
k_{\natural})$
along
$p^{\natural} \times q^{\natural}$,
i.e.\ the comma $\infty$-category
$(h_{\natural}p^{\natural} \downarrow k_{\natural}q^{\natural})$
of triples
$(a,b,\theta\colon hap \Rightarrow kbq)$.
As $pi = \mathrm{id}_{X}$ and $qi = f$, restriction along $i$ is a functor to
$(h_{\natural} \downarrow k_{\natural}f^{\natural})$
over
$\Hom(X,X') \times \Hom(Y,Y')$,
which on fibres is the equivalence \eqref{eq:comma-yoneda};
by \cref{lem:comma-fibrewise} it is an equivalence. The case of $j$ is dual.
Explicitly, for
$(h \downarrow k) = (f' \downarrow Y')$
in the first and
$(h \downarrow k) = (X' \downarrow g')$
in the second equivalence, a lax square
$\sigma\colon f'a \Rightarrow bf$,
resp.\
$\tau\colon ag \Rightarrow g'b$,
corresponds to
\begin{equation*}
c_{\sigma}\colon (u,v,\varphi) \mapsto (au,\,bv,\,b\varphi \circ \sigma u) \ ,
\quad \text{resp.} \quad d_{\tau}\colon (u,v,\psi) \mapsto (au,\,bv,\,\tau v
\circ a\psi) \ ,
\end{equation*}
as
$c_{\sigma}i = (a,bf,\sigma)$
and
$d_{\tau}j = (ag,b,\tau)$.
For
$(a,b) = (\mathrm{id}_{X},\mathrm{id}_{Y})$
and $(f',g') = (f,g)$, this shows that the $1$-morphisms
$(f \downarrow Y) \to (X \downarrow g)$
of $\Span_{\E}(X,Y)$ are the $e_{\eta}$, with
$e_{\eta}i = (\mathrm{id}_{X},f,\eta)$,
which is the first claim of (1); that the $1$-morphisms
$(X \downarrow g) \to (f \downarrow Y)$
of $\Span_{\E}(X,Y)$ are the
\begin{equation*}
d_{\varepsilon}\colon (u,v,\psi) \mapsto (u,\,v,\,\varepsilon v \circ f\psi) \ ,
\qquad d_{\varepsilon}j = (g,\mathrm{id}_{Y},\varepsilon) \ ,
\end{equation*}
for
$\varepsilon\colon fg \Rightarrow \mathrm{id}_{Y}$;
and that an endomorphism of $(f \downarrow Y)$, resp.\ $(X \downarrow g)$,
in $\Span_{\E}(X,Y)$ is the identity if and only if its restriction along $i$ is
$i$, resp.\ its restriction along $j$ is $j$.

\emph{Step 3: $e_{\eta}$ is an equivalence if and only if $\eta$ is a unit.} As
equivalences in $\Span_{\E}(X,Y)$ are detected in $\E$,
$e_{\eta}$ is an equivalence if and only if it has an inverse in
$\Span_{\E}(X,Y)$, i.e.\ by Step 2 an $\varepsilon$ with
\begin{equation*}
d_{\varepsilon}e_{\eta}i = (\mathrm{id}_{X},\,f,\,\varepsilon f \circ f\eta)
\simeq i \ , \qquad e_{\eta}d_{\varepsilon}j =
(g,\,\mathrm{id}_{Y},\,g\varepsilon \circ \eta g) \simeq j \ .
\end{equation*}
These are the triangle identities for $\eta$ and $\varepsilon$ in $\htwo\E$.
So $e_{\eta}$ is an equivalence if and only if $\eta$ is the unit of an
adjunction $f \dashv g$ \cite[Thm.~4.3.11]{riehlverity2016adjunctions}, and then
$e_{\eta}^{-1} = d_{\varepsilon}$
for its counit $\varepsilon$.

\emph{Step 4: adjunctions are transposing adjunctions.} Let $\mathrm{Ad}(f)$ be
the fibre of $\ell^{*}$ over $f$, the space of adjunctions $A$ with $A(l) = f$,
and let $\mathrm{Un}(f)$ be the space of pairs $(g,\eta)$ with $\eta$ the unit
of an adjunction $f \dashv g$. Both lie over $\mapp(Y,X)$, and
$A \mapsto (A(r),A(\eta))$
is a map
$\mathrm{Ad}(f) \to \mathrm{Un}(f)$
over $\mapp(Y,X)$. By Steps 2 and 3,
$\eta \mapsto e_{\eta}$
identifies the fibre of $\mathrm{Un}(f)$ over $g$ with the space of transposing
adjunctions between $f$ and $g$, so it suffices to show that
$\mathrm{Ad}(f) \to \mathrm{Un}(f)$
is an equivalence.
If $f$ is not a left adjoint, both spaces are empty.
Otherwise $\mathrm{Ad}(f)$ is contractible,
as $\ell^{*}$ is a monomorphism \cite[Thm.~4.4]{haugseng2021laxmonads},
and so is $\mathrm{Un}(f)$: let $f \dashv f^{R}$ with unit $\eta_{0}$.
The transposition
$\mapp(f^{R},g) \simeq \mapp(\mathrm{id}_{X},gf)$,
$\theta \mapsto \theta f \circ \eta_{0}$,
of
$(f^{R})^{\natural} \dashv f^{\natural}$
\eqref{eq:hom-adjunctions} is natural in $g \in \Hom(Y,X)$,
and by uniqueness of adjoints in $\htwo\E$ it identifies the units with the
invertible $\theta$. So $\mathrm{Un}(f)$ is the space of pairs
$(g,\theta\colon f^{R} \simeq g)$,
which is contractible.

\emph{Step 5: morphisms.} As $e$ and $e'$ are equivalences,
so are $e'_{\natural}$ and $e^{\natural}$, hence so are both projections from
$\Hom_{\TAdj(\E)}(e,e')$.
Composed with \eqref{eq:maps-of-commas},
they send $(c,d)$ to $\sigma$ and $\tau$ with $c \simeq c_{\sigma}$ and $d
\simeq d_{\tau}$; this gives the equivalences of (2).
By Step 3 we may take $e = e_{\eta}$ and $e' = e_{\eta'}$ for the units $\eta$,
$\eta'$ of $f \dashv g$ and $f' \dashv g'$, and then
$e^{-1} = d_{\varepsilon}$
for the counit $\varepsilon$ of $f \dashv g$. As $d \simeq e'ce^{-1}$,
the lax square $\tau$ is read off from
\begin{equation*}
dj \simeq e_{\eta'}c_{\sigma}d_{\varepsilon}j =
e_{\eta'}c_{\sigma}(g,\mathrm{id}_{Y},\varepsilon) =
e_{\eta'}(ag,\,b,\,b\varepsilon \circ \sigma g) = (ag,\,b,\,g'b\varepsilon \circ
g'\sigma g \circ \eta'ag) \ ,
\end{equation*}
so
$\tau = g'b\varepsilon \circ g'\sigma g \circ \eta'ag$,
the right mate of $\sigma$ (\cref{constr:mates}).
\end{proof}

\begin{rmk}
\label{rmk:comma-literature}
\gutentag{0022}%
Riehl--Shulman compare transposing adjunctions with adjunctions with a given
unit in their synthetic setting \cite[Cor.~11.21,
Thm.~11.23]{riehlshulman2017synthetic}. When $\E$ is presented by an
$\infty$-cosmos, \cref{thm:comma-adjunctions}(1) is
\cite[Prop.~4.1.1]{riehlverity2022elements},
with the strict comma objects of the cosmos;
for $\E = \Cat$ see \cite[Prop.~4.4.3]{riehlverity2015twocategory}.
If $\E$ does not have lax pullbacks, one can apply
\cref{thm:comma-adjunctions}(1) to $\Hom_{\E}(-,f)$ and $\Hom_{\E}(-,g)$ in
$\func(\E\opcat,\Cat)$, into which $\E$ embeds fully faithfully by the enriched
Yoneda lemma \cite{hinich2020yoneda,heine2023enriched}.
\cref{thm:comma-adjunctions}(2) upgrades \cref{lem:mate-correspondence}(1) from
spaces to $\infty$-categories. The equivalence
$(f'_{\natural} \downarrow f^{\natural}) \simeq (g^{\natural} \downarrow
g'_{\natural})$
itself only needs the adjunctions
$f'_{\natural} \dashv g'_{\natural}$
and
$g^{\natural} \dashv f^{\natural}$
of $\infty$-categories and \cref{lem:comma-fibrewise};
lax pullbacks realize it inside $\E$, as conjugation by $e$ and $e'$.
\end{rmk}

\begin{rmk}
\label{rmk:mates-alternative}
\gutentag{0023}%
\cref{lem:mates} also follows from the Riehl--Verity theorem
\cite[Thm.~4.4]{haugseng2021laxmonads} and the characterization of adjoints in
functor $(\infty,2)$-categories
\cite[Cor.~5.2.12]{abellangagnahaugseng2024straightening}:
a $1$-morphism of $\func(\A,\B)$ is a left adjoint if and only if its components
are left adjoints and its naturality squares are right Beck--Chevalley.
For $K \in \twoCat$, $\ell^{*}$ on $\mapp(K,-)$ is adjoint to
$\ell^{*}\colon \mapp(\Adj,\func(K,\E)) \to \mapp(\Delta^{1},\func(K,\E))$,
a monomorphism onto the left adjoints of $\func(K,\E)$ by
\cite[Thm.~4.4]{haugseng2021laxmonads}, and
\cite[Cor.~5.2.12]{abellangagnahaugseng2024straightening} computes these for $K
= \Delta^{0}$, $\Delta^{1}$, $\wtc$: for $K = \Delta^{1}$ they are the right
Beck--Chevalley squares between left adjoints,
and for $K = \wtc$ every $2$-morphism of $\func(\Delta^{1},\E)$ between two such
squares is a left adjoint of $\func(\wtc,\E)$.
So $\ell^{*}$ is a monomorphism and \eqref{eq:lff-square} is a pullback
(\cref{subsec:prelim-conventions}), i.e.\ $\ell^{*}$ is a locally full
inclusion.

When $\E$ has lax pullbacks, the two cases of
\cite[Cor.~5.2.12]{abellangagnahaugseng2024straightening} used here follow from
\cref{thm:comma-adjunctions}, granting that the hom-categories of $\func(K,\E)$
are the limits of the hom-categories of $\E$ indexed by $K$ \todo{Reference.}:
then lax pullbacks in $\func(K,\E)$ exist and are computed pointwise,
and a $1$-morphism of $\func(K,\E)$ is an equivalence if and only if its
components are. For $K = \Delta^{1}$, let
$L = (u,u',\varphi)\colon f_{0} \Rightarrow f_{1}$
be a $1$-morphism of $\func(\Delta^{1},\E)$ with $u$, $u'$ left adjoints,
and let $e$, $e'$ be the transposing adjunctions of $u \dashv u^{R}$ and $u'
\dashv u'^{R}$. If $\rmate{\varphi}$ is invertible, then
$R := (u^{R},u'^{R},\rmate{\varphi})\colon f_{1} \Rightarrow f_{0}$
is a $1$-morphism, and the morphism
$(c_{\varphi},d_{\rmate{\varphi}})\colon e \to e'$
of \cref{thm:comma-adjunctions}(2) makes $(e,e')$ a pointwise equivalence
$(L \downarrow f_{1}) \simeq (f_{0} \downarrow R)$
in
$\Span_{\func(\Delta^{1},\E)}(f_{0},f_{1})$,
i.e.\ a transposing adjunction between $L$ and $R$,
so $L \dashv R$ by \cref{thm:comma-adjunctions}(1). Conversely, if $L \dashv R$,
then $u \dashv R_{0}$ and $u' \dashv R_{1}$,
and the transposing adjunction of $L \dashv R$ is a $1$-morphism of $\TAdj(\E)$
between those of $u \dashv R_{0}$ and $u' \dashv R_{1}$,
with lax squares $\varphi$ and the naturality square of $R$.
By \cref{thm:comma-adjunctions}(2) the latter is $\rmate{\varphi}$,
which is therefore invertible. For $K = \wtc$ one argues in the same way,
transporting the identification of the pastings of the two squares along the
equivalences of \cref{thm:comma-adjunctions}(2).
\end{rmk}

\subsubsection*{Mates for reflections}

\gutentag{0024}%
Let
$c\colon \wtc \to \Delta^{1}$
collapse the $2$-morphism, and let
$\varepsilon,\eta\colon \wtc \to \Adj$
classify the counit
$lr \Rightarrow \mathrm{id}_{+}$
and the unit
$\mathrm{id}_{-} \Rightarrow rl$.
The \emph{walking reflection} and \emph{walking coreflection} are the pushouts
in $\twoCat$
\begin{equation*}
\Refl := \Adj \sqcup_{\wtc,\varepsilon} \Delta^{1} \ ,
\qquad \Corefl := \Adj \sqcup_{\wtc,\eta} \Delta^{1}
\end{equation*}
along $c$. We write $\ell$, $\rho$ also for the composites
$\Delta^{1} \to \Adj \to \Refl$,
$\Corefl$.

\begin{lem}[mates for reflections]
\label{lem:mates-refl}
\gutentag{0025}%
Let $\E$ be an $(\infty,2)$-category.
\begin{enumerate}
\item
$\func(\Refl,\E) \to \func(\Adj,\E)$
is fully faithful and exhibits $\func(\Refl,\E)$ as the full
sub-$(\infty,2)$-category on the reflections;
likewise $\func(\Corefl,\E)$ is the full sub-$(\infty,2)$-category on the
coreflections.
\item Hence for
$X \in \{\Refl,\Corefl\}$,
$\ell^{*}$ and
$\rho^{*}\colon \func(X,\E) \to \func(\Delta^{1},\E)$
are locally full inclusions with the $1$-morphisms described in
\cref{lem:mates}. The objects in the image of $\ell^{*}$, resp.\ $\rho^{*}$,
are the left, resp.\ right, adjoints of reflections for $X = \Refl$,
and of coreflections for $X = \Corefl$.
\end{enumerate}
\end{lem}

\begin{proof}
\gutentag{0026}%
(1) $\mapp(\wtc,\B)$ is the space of objects $x,y \in \B$ with a morphism of
$\Hom_{\B}(x,y)$, and $c^{*}$ sends $f$ to $\mathrm{id}_{f}$. As
$C^{\simeq} \to \func(\Delta^{1},C)^{\simeq}$
is a monomorphism onto the equivalences for every $\infty$-category $C$,
$c^{*}$ is a monomorphism onto the invertible $2$-morphisms.
Applying this to $\B = \func(K,\E)$, the map
\begin{equation*}
\mapp\bigl(K,\func(\Refl,\E)\bigr) \simeq \mapp\bigl(\Refl,\func(K,\E)\bigr)
\longrightarrow \mapp\bigl(\Adj,\func(K,\E)\bigr) \simeq
\mapp\bigl(K,\func(\Adj,\E)\bigr)
\end{equation*}
is a monomorphism onto the adjunctions in $\func(K,\E)$ with invertible counit.
A $2$-morphism of $\func(K,\E)$ is invertible if and only if its components at
the objects of $K$ are.\todo{Reference.}
So for $K = \Delta^{0}$, $\Delta^{1}$, $\wtc$ the image consists of all $K \to
\func(\Adj,\E)$ whose objects are reflections.
The case of $\Corefl$ is the same.
(2) follows from (1) and \cref{lem:mates}.
\end{proof}
