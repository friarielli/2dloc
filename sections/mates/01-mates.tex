\subsection{Lax squares and mates}
\label{subsec:mates-mates}

\begin{defi}
\label{def:lax-square}
\gutentag{000E}%
A \emph{lax square} in $\A$ is a diagram
\begin{equation*}
\begin{tikzcd}
A \ar[r, "f"] \ar[d, "g"'] & B \ar[d, "h"] \ar[dl, Rightarrow, shorten=3mm,
"\alpha"'] \\
C \ar[r, "k"'] & D
\end{tikzcd}
\end{equation*}
with a $2$-morphism
$\alpha\colon hf \Rightarrow kg$
from the top-right to the bottom-left composite.
It is a \emph{commutative square} if $\alpha$ is invertible.
\end{defi}

\gutentag{000F}%
The \emph{transpose} of a commutative square,
its reflection in the diagonal through $A$ and $D$,
is the commutative square with $g$, $k$ horizontal, $f$,
$h$ vertical and $2$-morphism
$\alpha^{-1}\colon kg \Rightarrow hf$.
So a commutative square has two underlying lax squares.

\gutentag{009C}%
\textbf{Important Convention:} we speak of the mates,
or of the Beck--Chevalley condition,
of a commutative square only after fixing one of its two underlying lax squares.
This way we can avoid saying ``horizontal/vertical left/right mate".
In particular, it could happen that we draw a lax square and take its mate,
while the lax square is in fact commutative.
Drawing a square as a lax square by no means implies that the square is not
commutative.

\begin{constru}[mates]
\label{constr:mates}
\gutentag{0010}%
Let $\alpha$ be the lax square of \cref{def:lax-square}.
\begin{enumerate}
\item If $f^{L} \dashv f$ and $k^{L} \dashv k$,
the \emph{left mate} of $\alpha$ is the lax square
\begin{equation*}
\begin{tikzcd}
B \ar[r, "h"] \ar[d, "f^{L}"'] & D \ar[d, "k^{L}"] \ar[dl, Rightarrow,
shorten=3mm, "\lmate{\alpha}"'] \\
A \ar[r, "g"'] & C
\end{tikzcd}
\qquad
\lmate{\alpha}\colon k^{L}h \xRightarrow{k^{L}h\eta_{f}} k^{L}hff^{L}
\xRightarrow{k^{L}\alpha f^{L}} k^{L}kgf^{L} \xRightarrow{\varepsilon_{k}gf^{L}}
gf^{L} \ .
\end{equation*}
\item If $g \dashv g^{R}$ and $h \dashv h^{R}$,
the \emph{right mate} of $\alpha$ is the lax square
\begin{equation*}
\begin{tikzcd}
C \ar[r, "g^{R}"] \ar[d, "k"'] & A \ar[d, "f"] \ar[dl, Rightarrow, shorten=3mm,
"\rmate{\alpha}"'] \\
D \ar[r, "h^{R}"'] & B
\end{tikzcd}
\qquad
\rmate{\alpha}\colon fg^{R} \xRightarrow{\eta_{h}fg^{R}} h^{R}hfg^{R}
\xRightarrow{h^{R}\alpha g^{R}} h^{R}kgg^{R} \xRightarrow{h^{R}k\varepsilon_{g}}
h^{R}k \ .
\end{equation*}
\end{enumerate}
Thus the left mate replaces the horizontal $1$-morphisms by their left adjoints
(rotating the square by $90^{\circ}$ to the left),
and the right mate replaces the vertical $1$-morphisms by their right adjoints
(rotating it to the right). Here $\eta$ and $\varepsilon$ denote units and
counits.
\end{constru}

\begin{defi}
\label{def:BC}
\gutentag{0011}%
A lax square is \emph{left Beck--Chevalley} if $f$,
$k$ admit left adjoints and $\lmate{\alpha}$ is invertible,
and \emph{right Beck--Chevalley} if $g$,
$h$ admit right adjoints and $\rmate{\alpha}$ is invertible.
\end{defi}

\begin{lem}[the two mates are conjugate]
\label{lem:mates-conjugate}
\gutentag{009D}%
Let $\alpha$ be the lax square of \cref{def:lax-square}, with $f^{L} \dashv f$,
$k^{L} \dashv k$, $g \dashv g^{R}$ and $h \dashv h^{R}$. Then
$\rmate{\alpha}\colon fg^{R} \Rightarrow h^{R}k$
is the right mate of
$\lmate{\alpha}\colon k^{L}h \Rightarrow gf^{L}$,
viewed as a lax square with identity horizontal edges and vertical edges $gf^{L}
\dashv fg^{R}$ and $k^{L}h \dashv h^{R}k$. In particular,
$\alpha$ is left Beck--Chevalley if and only if it is right Beck--Chevalley.
\end{lem}

\begin{proof}
\gutentag{009E}%
The right mate of $\lmate{\alpha}$ is the composite
\begin{equation*}
fg^{R} \xRightarrow{\ \eta\ } h^{R}k\,k^{L}h\,fg^{R}
\xRightarrow{\ h^{R}k\,\lmate{\alpha}\,fg^{R}\ } h^{R}k\,gf^{L}\,fg^{R}
\xRightarrow{\ \varepsilon\ } h^{R}k \ ,
\end{equation*}
where
$\eta = h^{R}\eta_{k}h \circ \eta_{h}$
is the unit of $k^{L}h \dashv h^{R}k$ and
$\varepsilon = \varepsilon_{g} \circ g\varepsilon_{f}g^{R}$
the counit of $gf^{L} \dashv fg^{R}$. Insert the formula of \cref{constr:mates}
for $\lmate{\alpha}$. \todo{2-categorical input:
check that the argument works verbatim.}
By interchange,
$\eta_{k}$ and $\varepsilon_{k}$ cancel by the triangle identity
$k\varepsilon_{k} \circ \eta_{k}k = \mathrm{id}_{k}$,
and $\eta_{f}$ and $\varepsilon_{f}$ by
$f\varepsilon_{f} \circ \eta_{f}f = \mathrm{id}_{f}$.
What remains is
$h^{R}k\varepsilon_{g} \circ h^{R}\alpha g^{R} \circ \eta_{h}fg^{R} =
\rmate{\alpha}$.
For the last claim: by \cref{lem:mate-correspondence}(1) and (3),
taking mates of squares with identity horizontal edges is a bijection compatible
with composition and identities, so it preserves and reflects invertibility.
\end{proof}

\begin{rmk}
\label{rmk:double-BC}
\gutentag{009F}%
The lemma does not extend to the \emph{double Beck--Chevalley map} of
Cnossen--Lenz--Linskens \cite[Def.~2.1(4)]{cnossenlenzlinskens2025universal}.
If $\alpha$ is left Beck--Chevalley, replacing $f$ and $k$ by their left
adjoints gives the \emph{mirror} of $\alpha$, the commutative square
\begin{equation*}
\begin{tikzcd}
B \ar[r, "f^{L}"] \ar[d, "h"'] & A \ar[d, "g"] \ar[dl, Rightarrow, shorten=3mm,
"{\lmate{\alpha}}^{-1}"'] \\
D \ar[r, "k^{L}"'] & C
\end{tikzcd}
\end{equation*}
whose right mate
$\delta\colon f^{L}h^{R} \Rightarrow g^{R}k^{L}$
is the double Beck--Chevalley map. Following
\cite{cnossenlenzlinskens2025universal},
$\alpha$ is \emph{biadjointable} if it is left Beck--Chevalley and $\delta$ is
invertible; we also say \emph{double Beck--Chevalley}.
This is a further condition: in $\Cat$, let
$u\colon \Delta^{1} \to \Delta^{2}$
be the inclusion of $\{0,2\}$, which is fully faithful with both adjoints,
and let $\alpha$ be the identity square with
$f = g = \mathrm{id}_{\Delta^{1}}$
and $h = k = u$. Its mates are the counit
$u^{L}u \simeq \mathrm{id}$
and the unit
$\mathrm{id} \simeq u^{R}u$,
both invertible, while
$\delta\colon u^{R} \Rightarrow u^{L}$
is at $1 \in \Delta^{2}$ the morphism $0 \to 1$ of $\Delta^{1}$.
\end{rmk}

\begin{lem}[mate correspondence]
\label{lem:mate-correspondence}
\gutentag{0012}%
In the situation of \cref{constr:mates}:
\begin{enumerate}
\item
$\alpha \mapsto \lmate{\alpha}$
and
$\alpha \mapsto \rmate{\alpha}$
are equivalences of spaces
\begin{equation*}
\mapp_{\Hom(B,C)}(k^{L}h,gf^{L}) \simeq \mapp_{\Hom(A,D)}(hf,kg) \simeq
\mapp_{\Hom(C,B)}(fg^{R},h^{R}k) \ .
\end{equation*}
\item They are mutually inverse: the right mate of $\lmate{\alpha}$ (with
respect to $f^{L} \dashv f$, $k^{L} \dashv k$) and the left mate of
$\rmate{\alpha}$ (with respect to $g \dashv g^{R}$,
$h \dashv h^{R}$) are $\alpha$.
\item Mates are compatible with horizontal and vertical pasting and with
identity squares.
\end{enumerate}
\end{lem}

\begin{proof}
\gutentag{0013}%
(1) The adjunctions
$(k^{L})_{\natural} \dashv k_{\natural}$
and
$f^{\natural} \dashv (f^{L})^{\natural}$
of \eqref{eq:hom-adjunctions} give
\begin{equation*}
\mapp(hf,kg) \simeq \mapp(k^{L}hf,g) \simeq \mapp(k^{L}h,gf^{L}) \ ,
\end{equation*}
and unwinding units and counits shows that the composite is
$\alpha \mapsto \lmate{\alpha}$.
Similarly
$h_{\natural} \dashv (h^{R})_{\natural}$
and
$(g^{R})^{\natural} \dashv g^{\natural}$
give
\begin{equation*}
\mapp(hf,kg) \simeq \mapp(f,h^{R}kg) \simeq \mapp(fg^{R},h^{R}k) \ .
\end{equation*}
(2) By the proof of (1),
$\alpha \mapsto \lmate{\alpha}$
is the inverse of the transposition along
$(k^{L})_{\natural} \dashv k_{\natural}$,
followed by the transposition along
$f^{\natural} \dashv (f^{L})^{\natural}$,
and, applied to the left mate, it shows that
$\beta \mapsto \rmate{\beta}$
is the transposition along
$(k^{L})_{\natural} \dashv k_{\natural}$,
followed by the inverse of the transposition along
$f^{\natural} \dashv (f^{L})^{\natural}$:
\begin{equation*}
\mapp(k^{L}h,gf^{L}) \simeq \mapp(h,kgf^{L}) \simeq \mapp(hf,kg) \ .
\end{equation*}
Transpositions along postcomposition adjunctions are natural with respect to
precomposition, and vice versa, so these two transpositions commute, and the
two composites are inverse. The same argument applies to the left mate of
$\rmate{\alpha}$.

(3) For identity squares, the left mate of
$\mathrm{id}\colon f \Rightarrow f$
(with $g$, $h$ identities and $k = f$) is
$\varepsilon_{f}f^{L} \circ f^{L}\eta_{f} \simeq \mathrm{id}_{f^{L}}$
by a triangle identity, and the left mate of a square with identity horizontal
$1$-morphisms is the square itself. Let $\alpha$ be pasted vertically with
$\alpha'\colon h'k \Rightarrow k'g'$,
where
$k'\colon C' \to D'$
has a left adjoint, giving
$P = \alpha'g \circ h'\alpha\colon h'hf \Rightarrow k'g'g$.
The left mate of $P$, and the horizontal pasting
$g'\lmate{\alpha} \circ \lmate{\alpha'}h$
of the left mates, are the transpositions along
$f^{\natural} \dashv (f^{L})^{\natural}$
of
$\varepsilon_{k'}g'g \circ k'^{L}P$
and of
\begin{equation*}
g'\varepsilon_{k}g \circ g'k^{L}\alpha \circ \varepsilon_{k'}g'k^{L}hf \circ
k'^{L}\alpha'k^{L}hf \circ k'^{L}h'\eta_{k}hf \ .
\end{equation*}
Moving $\varepsilon_{k'}$ to the left past $g'k^{L}\alpha$ and
$g'\varepsilon_{k}g$, and $\alpha'$ past $kk^{L}\alpha$ and $k\varepsilon_{k}g$,
by functoriality of horizontal composition, the latter becomes
\begin{equation*}
\varepsilon_{k'}g'g \circ k'^{L}\alpha'g \circ k'^{L}h'(k\varepsilon_{k}g \circ
kk^{L}\alpha \circ \eta_{k}hf) \ ,
\end{equation*}
and
$k\varepsilon_{k}g \circ kk^{L}\alpha \circ \eta_{k}hf \simeq k\varepsilon_{k}g
\circ \eta_{k}kg \circ \alpha \simeq \alpha$
by the same functoriality and a triangle identity. So the two agree.
Horizontal pasting is the same computation, with the composite adjunction
$f^{L}f'^{L} \dashv f'f$
of unit
$f'\eta_{f}f'^{L} \circ \eta_{f'}$
and counit
$\varepsilon_{f} \circ f^{L}\varepsilon_{f'}f$
(\cref{thm:adj-lifting}(2)). For right mates, use (2).
For $2$-categories, (2) and (3) are in \cite{kellystreet1974review}.
\end{proof}

\begin{rmk}
\label{rmk:mates-memo}
\gutentag{0014}%
If $\alpha$ is commutative with transpose $\alpha^{\mathrm{T}}$,
the left mate of $\alpha^{\mathrm{T}}$ replaces $g$,
$h$ by their left adjoints and its right mate replaces $f$,
$k$ by their right adjoints; by our convention these are mates of
$\alpha^{\mathrm{T}}$, not of $\alpha$.
\end{rmk}

\gutentag{00C3}%
We now recognize module maps over a lax idempotent monad $T$ on $\D$
(\cref{subsec:prelim-monads}) by mates, and prove \cref{thm:kz-modules}.
The \emph{unit square} of a $1$-morphism $f\colon x \to y$ of $\D$ is the lax
square
\begin{equation*}
\begin{tikzcd}
x \ar[r, "\eta_{x}"] \ar[d, "f"'] & Tx \ar[d, "Tf"] \ar[dl, Rightarrow,
shorten=3mm, "\alpha_{f}"'] \\
y \ar[r, "\eta_{y}"'] & Ty
\end{tikzcd}
\end{equation*}
whose $2$-morphism
$\alpha_{f}\colon Tf \circ \eta_{x} \simeq \eta_{y} \circ f$
is the naturality equivalence of $\eta$. If $\eta_{x}$ and $\eta_{y}$ are
reflective, with left adjoints $a$ and $b$, its left mate is a $2$-morphism
$\lmate{\alpha_{f}}\colon b \circ Tf \Rightarrow f \circ a$.
Let
$\D_{T} \subset \D$
be the locally full sub-$(\infty,2)$-category whose objects are the $x$ with
$\eta_{x}$ reflective, and whose $1$-morphisms are the $f$ with left
Beck--Chevalley unit square (\cref{def:BC}).
These classes are closed under equivalence and composition and contain the
identities by \cref{lem:mate-correspondence}(3), as the unit square of a
composite is the pasting of those of the factors.

\begin{thm}[recognition of module maps]
\label{thm:kz-module-maps}
\gutentag{00C4}%
Let $T$ be a lax idempotent monad on $\D$, and let $(x,a)$ and $(y,b)$ be
$T$-modules. A $1$-morphism $f\colon x \to y$ lifts to a module map $(x,a) \to
(y,b)$ if and only if its unit square is left Beck--Chevalley, and the space of
such lifts is then contractible, with structure equivalence
$\lmate{\alpha_{f}}\colon b \circ Tf \simeq f \circ a$.
More precisely, $U$ factors through an equivalence
$\modcat{T}{\D} \simeq \iota_{1}\D_{T}$.
Dually, for $T$ colax idempotent, $f$ lifts if and only if the transpose of its
unit square is right Beck--Chevalley.
\end{thm}

\begin{proof}[Proof of \cref{thm:kz-modules,thm:kz-module-maps}]
\gutentag{00C5}%
The colax case is the lax case for $\D^{\mathrm{co}}$
(\cref{def:kz}): $(-)^{\mathrm{co}}$ exchanges left and right
adjoints, and a lax square in $\D^{\mathrm{co}}$ is the transpose of a lax
square in $\D$, whose left mate in $\D^{\mathrm{co}}$ is its right mate in $\D$.
So let $T$ be lax idempotent, and let
$j\colon \iota_{1}\D_{T} \to \iota_{1}\D$
be the inclusion, a monomorphism of $\infty$-categories.
It suffices to show that $U \simeq j\Phi$ for an equivalence $\Phi$:
then the fibres of $U^{\simeq}$ are those of $j^{\simeq}$, contractible over
the objects of $\D_{T}$ and empty elsewhere, and the fibre of
$\mapp((x,a),(y,b)) \to \mapp(x,y)$
over $f$ is contractible if $f$ is in $\D_{T}$ and empty otherwise; the action
and the structure equivalence are identified by \cref{prop:kz-action} and
Step 1.

\emph{Step 1: $U$ factors through $j$.} For a module $(x,a)$, $\eta_{x}$ is
reflective with left adjoint $a$ (\cref{prop:kz-action}).
The structure $2$-morphism of a module map over $f$,
$b \circ Tf \simeq f \circ a$,
is $\lmate{\alpha_{f}}$ \cite[Lemma~10.6]{marmolejo1997doctrines}
(\cite[Cor.~6.3]{kellylack1997propertylike} for $2$-monads); so $f$ is in
$\D_{T}$. As $j$ is a monomorphism, $U = j\Phi$ for a unique $\Phi$.

\emph{Step 2: $j$ has the left adjoint $T$, with unit $\eta$.}
By Step 1 for the free modules $(Tz,\mu_{z})$ and the free maps $Tg$,
$T\colon \iota_{1}\D \to \iota_{1}\D$
factors through $j$. Let $y$ be in $\D_{T}$, with
$b \dashv \eta_{y}$.
Then $b\colon Ty \to y$ is in $\D_{T}$
\cite[Prop.~5.5]{marmolejo1997doctrines}
\todo{2-categorical input: check that \cite[Lemma~10.6,
Prop.~5.5]{marmolejo1997doctrines} work verbatim.},
and for $z \in \D$ restriction along
$\eta_{z}$,
$\mapp_{\D_{T}}(Tz,y) \to \mapp_{\D}(z,y)$,
has the inverse
$g \mapsto b \circ Tg$:
naturally in $g$ and in
$h \in \mapp_{\D_{T}}(Tz,y)$,
\begin{equation*}
b\,Tg\,\eta_{z} \simeq b\,\eta_{y}\,g \simeq g \ , \qquad b\,T(h\eta_{z})
\simeq h\,\mu_{z}\,T\eta_{z} \simeq h \ ,
\end{equation*}
by the counit of
$b \dashv \eta_{y}$,
the invertible left mate
$\lmate{\alpha_{h}}\colon b\,Th \Rightarrow h\,\mu_{z}$
(with
$\mu_{z} \dashv \eta_{Tz}$,
\cref{lem:kz-basic}(1)) and the unit law.
So $T \dashv j$ with unit $\eta$ \cite[Prop.~5.2.2.8]{htt}; the counit at $y$
is $b$, and the triangle identity
$b\eta_{y} \simeq \mathrm{id}_{y}$
is the counit of
$b \dashv \eta_{y}$.

\emph{Step 3: $j$-split simplicial objects.} Let $V$ be a simplicial object of
$\iota_{1}\D_{T}$ such that $jV$ is split, with colimit $x$, colimit cocone
$v_{n}\colon V_{n} \to x$
and
$s\colon x \to V_{0}$
with
$v_{0}s \simeq \mathrm{id}_{x}$
from the splitting. Every functor preserves this colimit
\cite[Rem.~4.7.2.4]{ha}, so $Tx$ is the colimit of $TV$, and the colimit of
$\eta_{V}\colon V \Rightarrow TV$
is $\eta_{x}$. By Step 2, applying $j$ to the counit of $T \dashv j$ gives a
natural transformation
$a_{\bullet}\colon TV \Rightarrow V$
of simplicial objects of $\iota_{1}\D$, with $a_{n}$ left adjoint to
$\eta_{V_{n}}$, and the counits $c_{n}$ of these adjunctions identify
$a_{\bullet}\eta_{V}$ with $\mathrm{id}_{V}$. Taking colimits gives
$a\colon Tx \to x$
with
$\psi\colon a\,Tv_{0} \simeq v_{0}\,a_{0}$
and
$c\colon a\eta_{x} \simeq \mathrm{id}_{x}$,
such that $cv_{0}$ is the composite
\begin{equation}
\label{eq:kz-colimit-counit}
a\,\eta_{x}\,v_{0} \overset{\alpha_{v_{0}}}{\simeq} a\,Tv_{0}\,\eta_{V_{0}}
\overset{\psi}{\simeq} v_{0}\,a_{0}\,\eta_{V_{0}} \overset{c_{0}}{\simeq} v_{0}
\ .
\end{equation}
\begin{enumerate}
\item[(i)] $a \dashv \eta_{x}$ with counit $c$ and unit
\begin{equation*}
\mathrm{id}_{Tx} \simeq Tv_{0}\,Ts \xRightarrow{\ Tv_{0}\,\theta_{0}\,Ts\ }
Tv_{0}\,\eta_{V_{0}}\,a_{0}\,Ts \overset{\alpha_{v_{0}}}{\simeq}
\eta_{x}\,v_{0}\,a_{0}\,Ts \overset{\psi}{\simeq} \eta_{x}\,a\,Tv_{0}\,Ts \simeq
\eta_{x}\,a \ ,
\end{equation*}
where $\theta_{0}$ is the unit of
$a_{0} \dashv \eta_{V_{0}}$:
by \eqref{eq:kz-colimit-counit} and interchange
\todo{2-categorical input: check that the argument works verbatim.},
the triangle identities reduce to those of
$a_{0} \dashv \eta_{V_{0}}$.
So $x$ is in $\D_{T}$.
\item[(ii)] The left mate of the unit square of $v_{0}$ is
\begin{equation*}
a\,Tv_{0} \xRightarrow{\ a\,Tv_{0}\,\theta_{0}\ }
a\,Tv_{0}\,\eta_{V_{0}}\,a_{0} \overset{\alpha_{v_{0}}}{\simeq}
a\,\eta_{x}\,v_{0}\,a_{0} \xRightarrow{\ c\,v_{0}\,a_{0}\ } v_{0}\,a_{0} \ ,
\end{equation*}
which is $\psi$ by \eqref{eq:kz-colimit-counit}, interchange and a triangle
identity. So $v_{0}$, and with it every $v_{n}$, is in $\D_{T}$.
\item[(iii)] Let
$g\colon x \to y$
with $y$ and $gv_{0}$ in $\D_{T}$. By \cref{lem:mate-correspondence}(3),
$\lmate{\alpha_{gv_{0}}}$
is the pasting of
$g\,\lmate{\alpha_{v_{0}}}$
and
$\lmate{\alpha_{g}}\,Tv_{0}$,
so
$\lmate{\alpha_{g}}\,Tv_{0}$
is invertible by (ii), and hence so is
$\lmate{\alpha_{g}} \simeq \lmate{\alpha_{g}}\,Tv_{0}\,Ts$.
So $g$ is in $\D_{T}$.
\end{enumerate}
As the mapping spaces of $\D_{T}$ are unions of components of those of $\D$,
(i)--(iii) show that $x$, with the cocone $v$, is a colimit of $V$ in
$\iota_{1}\D_{T}$, and $j$ preserves it.

\emph{Step 4: conclusion.} We apply \cite[Cor.~4.7.3.16]{ha} to $\Phi$ over
$\iota_{1}\D$. Its hypotheses hold: $U$ and $j$ have the left adjoints $F$ and
$T$ (Step 2); $U$-split simplicial objects have colimits preserved by $U$
(\cref{subsec:prelim-monads}), and $j$-split ones colimits preserved by $j$
(Step 3); $j$ is conservative, as for $f$ in $\D_{T}$ invertible in $\D$, the
left mates of the unit squares of $f$ and $f^{-1}$ paste to an identity
(\cref{lem:mate-correspondence}(3)), so
$\lmate{\alpha_{f^{-1}}}$
is invertible; and for $z \in \D$ the map
$Tz \to \Phi F(z) = Tz$
induced by $\eta_{z}$ is the identity. So $\Phi$ has a fully faithful left
adjoint, and $\Phi$ is an equivalence as $U$ is conservative.
\end{proof}

\begin{rmk}
\label{rmk:kz-literature}
\gutentag{00C6}%
By \cref{thm:kz-module-maps}, $\D_{T}$ refines $\modcat{T}{\D}$ to an
$(\infty,2)$-category, whose $2$-morphisms are all $2$-morphisms of $\D$ between
module maps; for $2$-monads, these are all compatible with the actions
\cite[Prop.~6.4]{kellylack1997propertylike}.
For $2$-categories, \cref{thm:kz-modules,thm:kz-module-maps} are due to Kock
\cite{kock1995monads} and Marmolejo \cite[Thm.~10.7]{marmolejo1997doctrines},
see also \cite[Thm.~6.2, Cor.~6.3]{kellylack1997propertylike}. We know of no
reference for $(\infty,2)$-categories; a homotopy coherent theory of lax
idempotent monads is announced by Abell\'an and Blom
\cite[Introduction]{abellan2026bifibrations}.
\end{rmk}
