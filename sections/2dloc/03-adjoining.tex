\subsection{Freely adjoining adjoints}
\label{subsec:2dloc-adjoining}

\begin{defi}
\label{def:adjoin-right}
Let $\D \in \tprl$ and let $S$ be a set of $1$-morphisms of $\D$, viewed as a map
$S\colon \bigoplus_{S}\Edge \to \D$ in $\tprl$.
\emph{Freely adjoining right adjoints to $S$} gives the pushout $\D\radj{S}$ in $\tprl$
\[
\begin{tikzcd}
\bigoplus_{S}\Edge \ar[r, "S"] \ar[d, "\bigoplus_{S}\Free(\ell)"'] & \D \ar[d, "L"] \\
\bigoplus_{S}\Free(\Adj) \ar[r] & \D\radj{S} \rlap{\ .}
\end{tikzcd}
\]
Dually, \emph{freely adjoining left adjoints} gives the pushout $\D\ladj{S}$ along $\bigoplus_{S}\Free(\rho)$.
For small $\A$ and a set $S$ of $1$-morphisms of $\A$, $\A\radj{S}$ and $\A\ladj{S}$ are the analogous pushouts in $\twoCat$ along $\ell$, resp.\ $\rho$.
\end{defi}

As $\Free$ preserves colimits, $\Free(\A\radj{S}) \simeq \Free(\A)\radj{y(S)}$.
The universal property of the pushout and \eqref{eq:free-univ} give:

\begin{prop}[universal property]
\label{prop:adjoin-univ}
For $\E \in \tprl$, restriction along $L\colon \D \to \D\radj{S}$ identifies $\func^{L}_{\Cat}(\D\radj{S},\E)$ with the sub-$(\infty,2)$-category of $\func^{L}_{\Cat}(\D,\E)$ whose
\begin{itemize}
\item objects are the $F$ such that $F(s)$ is a left adjoint for every $s \in S$;
\item $1$-morphisms are the $\alpha\colon F \Rightarrow G$ such that for every $(s\colon x \to y) \in S$ the transposed naturality square
\[
\begin{tikzcd}
F(x) \ar[r, "\alpha_{x}"] \ar[d, "F(s)"'] & G(x) \ar[d, "G(s)"] \ar[dl, Rightarrow, shorten=3mm, "\alpha_{s}^{-1}"'] \\
F(y) \ar[r, "\alpha_{y}"'] & G(y)
\end{tikzcd}
\qquad \alpha_{s}^{-1}\colon G(s)\alpha_{x} \simeq \alpha_{y}F(s) \ ,
\]
is right Beck--Chevalley, i.e.\ $\alpha_{x}F(s)^{R} \Rightarrow G(s)^{R}\alpha_{y}$ is invertible;
\item $2$-morphisms are all modifications.
\end{itemize}
The same holds for $\A\radj{S}$, with $\func(\A,\E)$ in place of $\func^{L}_{\Cat}(\D,\E)$, for every $(\infty,2)$-category $\E$.
For $\ladj{S}$, $F(s)$ must be a right adjoint, and the naturality square with $F(s)$, $G(s)$ horizontal and $2$-morphism $\alpha_{s}\colon \alpha_{y}F(s) \simeq G(s)\alpha_{x}$ must be left Beck--Chevalley.
\end{prop}

\begin{proof}
By \eqref{eq:free-univ}, $\func^{L}_{\Cat}(\D\radj{S},\E)$ is the pullback of $\func^{L}_{\Cat}(\D,\E) \to \prod_{S}\func(\Delta^{1},\E) \xleftarrow{\prod \ell^{*}} \prod_{S}\func(\Adj,\E)$. Apply \cref{lem:mates} below to the functor $\Delta^{1} \to \func^{L}_{\Cat}(\D,\E)$ classifying $F \Rightarrow G$, evaluated at $s$.
\end{proof}

\begin{rmk}
\label{rmk:adjoin-duality}
$(-)^{\mathrm{co}}$ exchanges left and right adjoints, and $\Adj^{\mathrm{co}} \simeq \Adj$ exchanges $\ell$ and $\rho$. Hence $\D\ladj{S} \simeq ((\D^{\mathrm{co}})\radj{S})^{\mathrm{co}}$, and we treat only $\radj{S}$.
\end{rmk}

\begin{exmp}
\label{ex:adjoin-basic}
\leavevmode
\begin{enumerate}
\item $\Edge\radj{e} \simeq \Free(\Adj)$ for the generic arrow $e$, and $(\Delta^{1})\radj{e} \simeq \Adj$.
\item For small $\A$ and $S$ all $1$-morphisms, $\A\radj{S}$ is the free $(\infty,2)$-category with right adjoints on $\A$. For $\A$ an $(\infty,1)$-category this is the $(\infty,2)$-category of spans of \cite{haugseng2018iteratedspans,cnossenlenzlinskens2025universal}; see \cref{rmk:2dloc-6ff}.
\todo{Check which of these references proves the universal property in this form.}
\item Let $\D = \Cat$, acting on itself by products, and $S = \{u\}$ for $u\colon X \to Y$. By \cref{thm:2dloc-local} below, $\Cat\radj{u}$ is the locally full sub-$(\infty,2)$-category of $\Cat$ on the $\C$ for which $u^{*}\colon \func(Y,\C) \to \func(X,\C)$ has a left adjoint $u_{!}$, and the $F\colon \C \to \C'$ for which the commutative square
\[
\begin{tikzcd}
\func(Y,\C) \ar[r, "u^{*}"] \ar[d, "F_{*}"'] & \func(X,\C) \ar[d, "F_{*}"] \ar[dl, Rightarrow, shorten=3mm, "="'] \\
\func(Y,\C') \ar[r, "u^{*}"'] & \func(X,\C')
\end{tikzcd}
\]
is left Beck--Chevalley, i.e.\ $u_{!}F_{*} \Rightarrow F_{*}u_{!}$ is invertible.
Here $u_{!}$ need not be pointwise: for $u\colon \{1\} \to \Delta^{1}$, a space $\C$ with two points is $u$-local, as $u^{*}$ is an equivalence, but pointwise left Kan extension along $u$ would require an initial object.
\end{enumerate}
\end{exmp}
