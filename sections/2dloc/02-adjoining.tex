\subsection{Formally adjoining reflections and coreflections}
\label{subsec:2dloc-adjoining}
\gutentag{003F}%

\begin{defi}
\label{def:adjoin}
\gutentag{0040}%
Let $\A$ be an $(\infty,2)$-category and $I$, $P$ classes of $1$-morphisms of $\A$.
Define $\A\lrrefl{I}{P}$ in $\twoCat$ to be the pushout  (enlarging the universe if necessary)
\[
\begin{tikzcd}[column sep=large]
\coprod_{I}\Delta^{1} \sqcup \coprod_{P}\Delta^{1} \ar[r, "{(I,P)}"] \ar[d, "\coprod_{I}\rho \sqcup \coprod_{P}\ell"'] & \A \ar[d, "L"] \\
\coprod_{I}\Refl \sqcup \coprod_{P}\Corefl \ar[r] & \A\lrrefl{I}{P} \rlap{\ .}
\end{tikzcd}
\]
We say that the functor $ L\colon\A \to \A\lrrefl{I}{P} $
\emph{formally maks $I$ reflective and $P$ coreflective}. 
We write $\A\refl{I} := \A\lrrefl{I}{\emptyset}$ and $\A\corefl{P} := \A\lrrefl{\emptyset}{P}$.
\end{defi}

\gutentag{0041}%
So $Li$ acquires a left adjoint with invertible counit, and $Lp$ a right adjoint with invertible unit.
We will compare this with freely adjoining plain adjoints in \cref{subsec:2dloc-plain}.
Pasting pushouts gives $\A\lrrefl{I}{P} \simeq \A\refl{I}\corefl{L_{I}(P)} \simeq \A\corefl{P}\refl{L_{P}(I)}$, where $L_{I}\colon \A \to \A\refl{I}$ and $L_{P}\colon \A \to \A\corefl{P}$; the corresponding statement on local objects will be \cref{prop:2dloc-commute}.

\begin{prop}[universal property]
\label{prop:adjoin-univ}
\gutentag{0042}%
For every $(\infty,2)$-category $\E$, restriction along $L\colon \A \to \A\lrrefl{I}{P}$ identifies $\func(\A\lrrefl{I}{P},\E)$ with the locally full sub-$(\infty,2)$-category\reviewcomment{We said locally full so description of $ 2 $-morphisms is not needed.} of $\func(\A,\E)$ whose
\begin{itemize}
\item objects are the $F$ such that $F(i)$ is reflective for every $i \in I$ and $F(p)$ is coreflective for every $p \in P$;
\item $1$-morphisms are the $\alpha\colon F \Rightarrow G$ such that for every $(p\colon x \to y) \in P$ the transposed naturality square
\[
\begin{tikzcd}
F(x) \ar[r, "\alpha_{x}"] \ar[d, "F(p)"'] & G(x) \ar[d, "G(p)"] \ar[dl, Rightarrow, shorten=3mm, "\alpha_{p}^{-1}"'] \\
F(y) \ar[r, "\alpha_{y}"'] & G(y)
\end{tikzcd}
\qquad \alpha_{p}^{-1}\colon G(p)\alpha_{x} \simeq \alpha_{y}F(p) \ ,
\]
is right Beck--Chevalley, i.e.\ $\alpha_{x}F(p)^{R} \Rightarrow G(p)^{R}\alpha_{y}$ is invertible, and for every $(i\colon x \to y) \in I$ the naturality square
\[
\begin{tikzcd}
F(x) \ar[r, "F(i)"] \ar[d, "\alpha_{x}"'] & F(y) \ar[d, "\alpha_{y}"] \ar[dl, Rightarrow, shorten=3mm, "\alpha_{i}"'] \\
G(x) \ar[r, "G(i)"'] & G(y)
\end{tikzcd}
\qquad \alpha_{i}\colon \alpha_{y}F(i) \simeq G(i)\alpha_{x} \ ,
\]
is left Beck--Chevalley, i.e.\ $G(i)^{L}\alpha_{y} \Rightarrow \alpha_{x}F(i)^{L}$ is invertible.
\end{itemize}
\end{prop}

\begin{proof}
\gutentag{0043}%
$\func(-,\E)$ turns the pushout into the pullback of
\[
\func(\A,\E) \to \textstyle\prod_{I \sqcup P}\func(\Delta^{1},\E) \xleftarrow{\prod_{I}\rho^{*} \times \prod_{P}\ell^{*}} \prod_{I}\func(\Refl,\E) \times \prod_{P}\func(\Corefl,\E) \ .
\]
By \cref{lem:mates-refl} the right-hand map is a locally full inclusion, hence so is its pullback (\cref{subsec:prelim-conventions}). Apply \cref{lem:mates-refl} to the functor $\Delta^{1} \to \func(\A,\E)$ classifying $F \Rightarrow G$, evaluated at $p$, resp.\ $i$.
\end{proof}

\begin{rmk}
\label{rmk:lax-epi}
\gutentag{0044}%
Following \cite{adamekelbashirsobralvelebil2001laxepi}, a functor $j\colon \A \to \B$ is a \emph{lax epimorphism} if every $j^{*}\colon \func(\B,\E) \to \func(\A,\E)$ is locally fully faithful, and an \emph{epimorphism} if every $j^{*}$ is fully faithful. By \cref{lem:mates-refl}, $\rho\colon \Delta^{1} \to \Refl$ and $\ell\colon \Delta^{1} \to \Corefl$ are lax epimorphisms but not epimorphisms, the difference being the Beck--Chevalley condition on $1$-morphisms.
Locally fully faithful functors are stable under products and pullbacks (\cref{subsec:prelim-conventions}). As $\func(-,\E)$ turns pushouts into pullbacks, $\A \to \A\lrrefl{I}{P}$ is a lax epimorphism.
\end{rmk}

\begin{rmk}
\label{rmk:adjoin-duality}
\gutentag{0045}%
$(-)^{\mathrm{co}}$ exchanges left and right adjoints as well as units and counits, so $\Adj^{\mathrm{co}} \simeq \Adj$ induces $\Refl^{\mathrm{co}} \simeq \Corefl$, exchanging $\ell$ and $\rho$. Hence $\A\lrrefl{I}{P} \simeq ((\A^{\mathrm{co}})\lrrefl{P}{I})^{\mathrm{co}}$.
\end{rmk}

\begin{exmp}
\label{ex:adjoin-basic}
\gutentag{0046}%
$(\Delta^{1})\corefl{e} \simeq \Corefl$ for the generic arrow $e$.
In general $\A\lrrefl{I}{P}$ is a free construction and not a sub-$(\infty,2)$-category of $\A$. In \cref{subsec:2dloc-local} we will describe the local objects inside a given $(\infty,2)$-category, and \cref{thm:2dloc-local} will identify the two in the presentable case
\reviewcomment{Identify them if the left adjoint exists?}. For small $\A$ and sets $I$, $P$ this will give $\Free(\A\lrrefl{I}{P}) \simeq \Pc_{y(I),y(P)} \subset \Free(\A)$ (\cref{rmk:free-local}).
\end{exmp}
