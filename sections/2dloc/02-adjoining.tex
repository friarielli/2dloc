\subsection{Freely adjoining adjoints}
\label{subsec:2dloc-adjoining}

\begin{defi}
\label{def:adjoin}
Let $\D \in \tprl$ and let $I$, $P$ be sets of $1$-morphisms of $\D$, viewed as maps
$I\colon \bigoplus_{I}\Edge \to \D$ and $P\colon \bigoplus_{P}\Edge \to \D$ in $\tprl$.
\emph{Freely adjoining left adjoints to $I$ and right adjoints to $P$} gives the pushout $\D\lradj{I}{P}$ in $\tprl$
\[
\begin{tikzcd}[column sep=large]
\bigoplus_{I}\Edge \oplus \bigoplus_{P}\Edge \ar[r, "{(I,P)}"] \ar[d, "\bigoplus_{I}\Free(\rho) \oplus \bigoplus_{P}\Free(\ell)"'] & \D \ar[d, "L"] \\
\bigoplus_{I}\Free(\Adj) \oplus \bigoplus_{P}\Free(\Adj) \ar[r] & \D\lradj{I}{P} \rlap{\ .}
\end{tikzcd}
\]
We write $\D\ladj{I} := \D\lradj{I}{\emptyset}$ and $\D\radj{P} := \D\lradj{\emptyset}{P}$.
For small $\A$ and sets $I$, $P$ of $1$-morphisms of $\A$, $\A\lradj{I}{P}$ is the analogous pushout in $\twoCat$ along $\rho$ on the $I$-summands and $\ell$ on the $P$-summands.
\end{defi}

As $\Free$ preserves colimits, $\Free(\A\lradj{I}{P}) \simeq \Free(\A)\lradj{y(I)}{y(P)}$.
Pasting pushouts gives $\D\lradj{I}{P} \simeq \D\ladj{I}\radj{L_{I}(P)} \simeq \D\radj{P}\ladj{L_{P}(I)}$, where $L_{I}\colon \D \to \D\ladj{I}$ and $L_{P}\colon \D \to \D\radj{P}$; see \cref{prop:2dloc-commute} for the corresponding statement on local objects.
The universal property of the pushout and \eqref{eq:free-univ} give:

\begin{prop}[universal property]
\label{prop:adjoin-univ}
For $\E \in \tprl$, restriction along $L\colon \D \to \D\lradj{I}{P}$ identifies $\func^{L}_{\Cat}(\D\lradj{I}{P},\E)$ with the sub-$(\infty,2)$-category of $\func^{L}_{\Cat}(\D,\E)$ whose
\begin{itemize}
\item objects are the $F$ such that $F(i)$ is a right adjoint for every $i \in I$ and $F(p)$ is a left adjoint for every $p \in P$;
\item $1$-morphisms are the $\alpha\colon F \Rightarrow G$ such that for every $(p\colon x \to y) \in P$ the transposed naturality square
\[
\begin{tikzcd}
F(x) \ar[r, "\alpha_{x}"] \ar[d, "F(p)"'] & G(x) \ar[d, "G(p)"] \ar[dl, Rightarrow, shorten=3mm, "\alpha_{p}^{-1}"'] \\
F(y) \ar[r, "\alpha_{y}"'] & G(y)
\end{tikzcd}
\qquad \alpha_{p}^{-1}\colon G(p)\alpha_{x} \simeq \alpha_{y}F(p) \ ,
\]
is right Beck--Chevalley, i.e.\ $\alpha_{x}F(p)^{R} \Rightarrow G(p)^{R}\alpha_{y}$ is invertible, and for every $(i\colon x \to y) \in I$ the naturality square
\[
\begin{tikzcd}
F(x) \ar[r, "F(i)"] \ar[d, "\alpha_{x}"'] & F(y) \ar[d, "\alpha_{y}"] \ar[dl, Rightarrow, shorten=3mm, "\alpha_{i}"'] \\
G(x) \ar[r, "G(i)"'] & G(y)
\end{tikzcd}
\qquad \alpha_{i}\colon \alpha_{y}F(i) \simeq G(i)\alpha_{x} \ ,
\]
is left Beck--Chevalley, i.e.\ $G(i)^{L}\alpha_{y} \Rightarrow \alpha_{x}F(i)^{L}$ is invertible;
\item $2$-morphisms are all modifications.
\end{itemize}
The same holds for $\A\lradj{I}{P}$, with $\func(\A,\E)$ in place of $\func^{L}_{\Cat}(\D,\E)$, for every $(\infty,2)$-category $\E$.
\end{prop}

\begin{proof}
By \eqref{eq:free-univ}, $\func^{L}_{\Cat}(\D\lradj{I}{P},\E)$ is the pullback of
\[
\func^{L}_{\Cat}(\D,\E) \to \textstyle\prod_{I \sqcup P}\func(\Delta^{1},\E) \xleftarrow{\prod_{I}\rho^{*} \times \prod_{P}\ell^{*}} \prod_{I \sqcup P}\func(\Adj,\E) \ .
\]
Apply \cref{lem:mates} to the functor $\Delta^{1} \to \func^{L}_{\Cat}(\D,\E)$ classifying $F \Rightarrow G$, evaluated at $p$, resp.\ $i$.
\end{proof}

\begin{rmk}
\label{rmk:lax-epi}
Following \cite{adamekelbashirsobralvelebil2001laxepi}, a functor $j\colon \A \to \B$ is a \emph{lax epimorphism} if every $j^{*}\colon \func(\B,\E) \to \func(\A,\E)$ is locally fully faithful, and an \emph{epimorphism} if every $j^{*}$ is fully faithful. By \cref{lem:mates}, $\ell$ and $\rho$ are lax epimorphisms but not epimorphisms, the difference being the Beck--Chevalley condition on $1$-morphisms.
Since $\func(-,\E)$ turns pushouts into pullbacks and locally full sub-$(\infty,2)$-categories are stable under pullback, $\A \to \A\lradj{I}{P}$ is a lax epimorphism, and $L^{*}\colon \func^{L}_{\Cat}(\D\lradj{I}{P},\E) \to \func^{L}_{\Cat}(\D,\E)$ is a locally full sub-$(\infty,2)$-category for every $\E \in \tprl$.
\end{rmk}

\begin{rmk}
\label{rmk:adjoin-duality}
$(-)^{\mathrm{co}}$ exchanges left and right adjoints, and $\Adj^{\mathrm{co}} \simeq \Adj$ exchanges $\ell$ and $\rho$. Hence $\D\lradj{I}{P} \simeq ((\D^{\mathrm{co}})\lradj{P}{I})^{\mathrm{co}}$.
\end{rmk}

\begin{exmp}
\label{ex:adjoin-basic}
\leavevmode
\begin{enumerate}
\item $\Edge\radj{e} \simeq \Free(\Adj)$ for the generic arrow $e$, and $(\Delta^{1})\radj{e} \simeq \Adj$.
\item For small $\A$ and $P$ all $1$-morphisms, $\A\radj{P}$ is the free $(\infty,2)$-category with right adjoints on $\A$. For $\A$ an $(\infty,1)$-category this is the $(\infty,2)$-category of spans of \cite{haugseng2018iteratedspans,cnossenlenzlinskens2025universal}.
\todo{Check which of these references proves the universal property in this form.}
\item Let $\D = \Cat$, acting on itself by products, and $P = \{u\}$ for $u\colon X \to Y$. By \cref{thm:2dloc-local} below, $\Cat\radj{u}$ is the locally full sub-$(\infty,2)$-category of $\Cat$ on the $\C$ for which $u^{*}\colon \func(Y,\C) \to \func(X,\C)$ has a left adjoint $u_{!}$, and the $F\colon \C \to \C'$ for which the commutative square
\[
\begin{tikzcd}
\func(Y,\C) \ar[r, "u^{*}"] \ar[d, "F_{*}"'] & \func(X,\C) \ar[d, "F_{*}"] \ar[dl, Rightarrow, shorten=3mm, "="'] \\
\func(Y,\C') \ar[r, "u^{*}"'] & \func(X,\C')
\end{tikzcd}
\]
is left Beck--Chevalley, i.e.\ $u_{!}F_{*} \Rightarrow F_{*}u_{!}$ is invertible.
Here $u_{!}$ need not be pointwise: for $u\colon \{1\} \to \Delta^{1}$, a space $\C$ with two points lies in $\Cat\radj{u}$, as $u^{*}$ is an equivalence, but pointwise left Kan extension along $u$ would require an initial object.
Dually, $\Cat\ladj{u}$ consists of the $\C$ for which $u^{*}$ has a right adjoint $u_{*}$, and the $F$ with $F_{*}u_{*} \Rightarrow u_{*}F_{*}$ invertible.
\item Let $\D = \Cat$ and $I = P = \{\nabla\}$ for the fold map $\nabla\colon \mathrm{pt} \sqcup \mathrm{pt} \to \mathrm{pt}$, so that $\nabla^{*}\colon \C \to \C \times \C$ is the diagonal. Then $\Cat\lradj{\nabla}{\nabla}$ consists of the $\infty$-categories with binary products and binary coproducts, and the functors preserving both.
\end{enumerate}
\end{exmp}
