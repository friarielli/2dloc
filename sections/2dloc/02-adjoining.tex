\subsection{Formally adjoining reflections and coreflections}
\label{subsec:2dloc-adjoining}

\begin{defi}
\label{def:adjoin}
\gutentag{0031}%
Let $\A$ be an $(\infty,2)$-category and $I$,
$P \subset \func(\Delta^{1},\A)$
locally full sub-$(\infty,2)$-categories. We draw a $1$-morphism
$\sigma\colon i \to i'$
of $I$, a commutative square, with $i$ and $i'$ horizontal, and a $1$-morphism
$\sigma\colon p \to p'$
of $P$ with $p$ and $p'$ vertical, as in \cref{lem:mates}(2), resp.\ (1).
Let $Q$ be a class of commutative squares $\sigma$ in $I \cap P$,
i.e.\ drawn as in \cref{def:lax-square}, $\sigma$ is a $1$-morphism $f \to k$
of $I$ and a $1$-morphism $g \to h$ of $P$.
Define $\A\lrqrefl{I}{P}{Q}$ in $\twoCat$ to be the pushout (enlarging the
universe if necessary)
\begin{equation*}
\begin{tikzcd}[column sep=large]
(\Delta^{1} \times I) \sqcup (\Delta^{1} \times P) \sqcup
\coprod_{Q}(\Delta^{1} \times \Delta^{1}) \ar[r, "\mathrm{ev}"] \ar[d,
"(\rho \times \mathrm{id}) \sqcup (\ell \times \mathrm{id}) \sqcup
\coprod_{Q}(\rho \times \ell)"'] & \A \ar[d, "\gamma"] \\
(\Refl \times I) \sqcup (\Corefl \times P) \sqcup
\coprod_{Q}(\Refl \times \Corefl) \ar[r] & \A\lrqrefl{I}{P}{Q} \rlap{\ ,}
\end{tikzcd}
\end{equation*}
where $\mathrm{ev}$ is evaluation, and
$\Delta^{1} \times \Delta^{1}$
is drawn with the first factor horizontal. We say that
$\gamma\colon \A \to \A\lrqrefl{I}{P}{Q}$
\emph{formally makes $I$ reflective, $P$ coreflective and $Q$ double
Beck--Chevalley}, and write
$\A\lrrefl{I}{P} := \A\lrqrefl{I}{P}{\emptyset}$,
$\A\refl{I} := \A\lrrefl{I}{\emptyset}$
and
$\A\corefl{P} := \A\lrrefl{\emptyset}{P}$.
\end{defi}

\gutentag{00A4}%
A class $S$ of $1$-morphisms of $\A$ is regarded as the locally full
sub-$(\infty,2)$-category of $\func(\Delta^{1},\A)$ on $S$ whose $1$-morphisms
are the equivalences. For classes $I$ and $P$,
the pushout above is equivalent to the pushout along
$\coprod_{I}\rho \sqcup \coprod_{P}\ell\colon \coprod_{I \sqcup P}\Delta^{1} \to
\coprod_{I}\Refl \sqcup \coprod_{P}\Corefl$,
since both have the universal property of \cref{prop:adjoin-univ}:
the images of $\rho^{*}$ and $\ell^{*}$ (\cref{lem:mates-refl}) contain all
equivalences between their objects, so equivalences impose no condition.
In the same way, the pushout along a functor
$K \to \func(\Delta^{1},\A)$
from any $(\infty,2)$-category $K$ only depends on the locally full
sub-$(\infty,2)$-category generated by the images of the objects and
$1$-morphisms of $K$.
Similarly, $Q$ is a class and not a locally full sub-$(\infty,2)$-category of
$\func(\Delta^{1} \times \Delta^{1},\A)$:
by \cref{lem:biadjointable}(2),
a $1$-morphism between squares of $Q$, a cube, would only impose that its four
lateral faces become Beck--Chevalley, which is achieved by adding these faces to
$I$ and $P$.
Nor is the assumption $Q \subset I \cap P$ a restriction:
the proof of \cref{prop:adjoin-univ} will show that adding the edges of the
squares of $Q$ to $I$ and $P$, and the squares themselves as $1$-morphisms,
does not change $\A\lrqrefl{I}{P}{Q}$.

\gutentag{0032}%
So $\gamma(i)$ acquires a left adjoint with invertible counit,
$\gamma(p)$ a right adjoint with invertible unit,
the $1$-morphisms of $I$ and $P$ become left, resp.\ right,
Beck--Chevalley squares, and the squares of $Q$ become double Beck--Chevalley
(\cref{prop:adjoin-univ}).
We will compare this with freely adjoining plain adjoints in
\cref{subsec:2dloc-plain}.
For $Q = \emptyset$, pasting pushouts gives
$\A\lrrefl{I}{P} \simeq \A\refl{I}\corefl{\gamma_{I}(P)} \simeq
\A\corefl{P}\refl{\gamma_{P}(I)}$,
where
$\gamma_{I}\colon \A \to \A\refl{I}$
and
$\gamma_{P}\colon \A \to \A\corefl{P}$,
and
$\gamma_{I}(P) \subset \func(\Delta^{1},\A\refl{I})$
is the locally full sub-$(\infty,2)$-category generated by the image of $P$,
likewise $\gamma_{P}(I)$; the corresponding statement on local objects will be
\cref{prop:2dloc-commute}. The second step can be taken along any locally full
$P' \subset \func(\Delta^{1},\A\refl{I})$,
which may contain squares that do not come from $\A$,
as we will see in \cref{rmk:biadjointable}; we write
$\A\refl{I}\corefl{P'}$
for the result. \Cref{rmk:biadjointable} will also express
$\A\lrqrefl{I}{P}{Q}$
in this form.

\begin{prop}[universal property]
\label{prop:adjoin-univ}
\gutentag{0033}%
For every $(\infty,2)$-category $\E$, restriction along
$\gamma\colon \A \to \A\lrqrefl{I}{P}{Q}$
identifies
$\func(\A\lrqrefl{I}{P}{Q},\E)$
with the locally full sub-$(\infty,2)$-category of $\func(\A,\E)$ (so determined
by its objects and $1$-morphisms) whose
\begin{itemize}
\item objects are the $F$ such that $F(i)$ is reflective and $F(\sigma)$ is left
Beck--Chevalley for every object $i$ and $1$-morphism $\sigma$ of $I$,
and $F(p)$ is coreflective and $F(\sigma)$ is right Beck--Chevalley for every
object $p$ and $1$-morphism $\sigma$ of $P$,
the squares being drawn as in \cref{def:adjoin},
and $F(\sigma)$ is double Beck--Chevalley (\cref{rmk:double-BC}) for every
$\sigma$ in $Q$;
\item $1$-morphisms are the
$\alpha\colon F \Rightarrow G$
such that for every object $p\colon x \to y$ of $P$ the transposed naturality
square
\begin{equation*}
\begin{tikzcd}
F(x) \ar[r, "\alpha_{x}"] \ar[d, "F(p)"'] & G(x) \ar[d, "G(p)"] \ar[dl,
Rightarrow, shorten=3mm, "\alpha_{p}^{-1}"'] \\
F(y) \ar[r, "\alpha_{y}"'] & G(y)
\end{tikzcd}
\qquad \alpha_{p}^{-1}\colon G(p)\alpha_{x} \simeq \alpha_{y}F(p) \ ,
\end{equation*}
is right Beck--Chevalley, i.e.\
$\alpha_{x}F(p)^{R} \Rightarrow G(p)^{R}\alpha_{y}$
is invertible, and for every object $i\colon x \to y$ of $I$ the naturality
square
\begin{equation*}
\begin{tikzcd}
F(x) \ar[r, "F(i)"] \ar[d, "\alpha_{x}"'] & F(y) \ar[d, "\alpha_{y}"] \ar[dl,
Rightarrow, shorten=3mm, "\alpha_{i}"'] \\
G(x) \ar[r, "G(i)"'] & G(y)
\end{tikzcd}
\qquad \alpha_{i}\colon \alpha_{y}F(i) \simeq G(i)\alpha_{x} \ ,
\end{equation*}
is left Beck--Chevalley, i.e.\
$G(i)^{L}\alpha_{y} \Rightarrow \alpha_{x}F(i)^{L}$
is invertible.
\end{itemize}
In particular $Q$ imposes no condition on $1$-morphisms.
\end{prop}

\begin{proof}
\gutentag{0034}%
$\func(-,\E)$ turns the pushout into the pullback of
\begin{equation*}
\begin{gathered}
\func(\A,\E) \to \func\bigl(I,\func(\Delta^{1},\E)\bigr) \times
\func\bigl(P,\func(\Delta^{1},\E)\bigr) \times \prod_{Q}\func(\Delta^{1} \times
\Delta^{1},\E) \\
\xleftarrow{\ (\rho^{*})_{*} \times (\ell^{*})_{*} \times \prod_{Q}(\rho \times
\ell)^{*}\ } \func\bigl(I,\func(\Refl,\E)\bigr) \times
\func\bigl(P,\func(\Corefl,\E)\bigr) \times \prod_{Q}\func(\Refl \times
\Corefl,\E) \ .
\end{gathered}
\end{equation*}
By \cref{lem:mates-refl}, \cref{lem:biadjointable}(2) and
\cref{subsec:prelim-conventions} the right-hand map is a locally full
inclusion, hence so is its pullback.
By \cite[Cor.~2.6.6]{abellangagnahaugseng2024straightening},
$F$ lies in its image if and only if $F \circ \mathrm{ev}$, viewed as a functor
$I \to \func(\Delta^{1},\E)$,
sends the objects and $1$-morphisms of $I$ into the image of $\rho^{*}$,
and likewise for $P$ and $\ell^{*}$; by \cref{lem:mates-refl}(2) and
\cref{lem:mates} these are the conditions on objects.
Likewise $\alpha$ lies in the image if and only if the functor
$\Delta^{1} \times I \to \func(\Delta^{1},\E)$
it induces sends the $1$-morphisms of $\Delta^{1} \times I$ into the image of
$\rho^{*}$, and likewise for $P$. These $1$-morphisms are composites of the
$(0 \to 1,\mathrm{id}_{i})$,
which go to the naturality squares at the objects $i$ of $I$,
and of the $(\mathrm{id},\sigma)$, which go to $F(\sigma)$ and $G(\sigma)$.
As Beck--Chevalley squares are closed under pasting
(\cref{lem:mate-correspondence}(3)), only the naturality squares impose a
condition, which is the stated one by \cref{lem:mates-refl}(2) and
\cref{lem:mates}.
For $\sigma$ in $Q$, by \cref{lem:biadjointable}(2),
$F$ satisfies the condition at $\sigma$ if and only if $F(\sigma)$ is
biadjointable, i.e.\ double Beck--Chevalley,
its edges being reflective, resp.\ coreflective, already;
and $\alpha$ does if and only if its naturality squares at the horizontal edges
of $\sigma$ are left, and at the vertical ones right, Beck--Chevalley.
As these edges are objects of $I$ and $P$, this is already implied.
\end{proof}

\begin{rmk}
\label{rmk:lax-epi}
\gutentag{0035}%
Following \cite{adamekelbashirsobralvelebil2001laxepi},
a functor $j\colon \A \to \B$ is a \emph{lax epimorphism} if every
$j^{*}\colon \func(\B,\E) \to \func(\A,\E)$
is locally fully faithful, and an \emph{epimorphism} if every $j^{*}$ is fully
faithful. By \cref{lem:mates-refl},
$\rho\colon \Delta^{1} \to \Refl$
and
$\ell\colon \Delta^{1} \to \Corefl$
are lax epimorphisms but not epimorphisms,
the difference being the Beck--Chevalley condition on $1$-morphisms.
By the proof of \cref{prop:adjoin-univ},
$\gamma\colon \A \to \A\lrqrefl{I}{P}{Q}$
is a lax epimorphism.
The squares of $I$, $P$ and $Q$ only add conditions on objects:
for the classes $I_{0}$, $P_{0}$ of objects of $I$, $P$, the functor
$\A\lrrefl{I_{0}}{P_{0}} \to \A\lrqrefl{I}{P}{Q}$
induced by $\gamma$ is an epimorphism, since by \cref{prop:adjoin-univ}
restriction along it is the inclusion of a locally full
sub-$(\infty,2)$-category with the same $1$-morphisms between its objects,
i.e.\ of a full one.
\end{rmk}

\begin{rmk}
\label{rmk:adjoin-duality}
\gutentag{0036}%
$(-)^{\mathrm{co}}$ exchanges left and right adjoints as well as units and
counits, so
$\Adj^{\mathrm{co}} \simeq \Adj$
induces
$\Refl^{\mathrm{co}} \simeq \Corefl$,
exchanging $\ell$ and $\rho$. As
$\func(\Delta^{1},\A^{\mathrm{co}}) \simeq \func(\Delta^{1},\A)^{\mathrm{co}}$,
we get
$\A\lrqrefl{I}{P}{Q} \simeq
((\A^{\mathrm{co}})\lrqrefl{P^{\mathrm{co}}}{I^{\mathrm{co}}}{Q^{\mathrm{co}}})^{\mathrm{co}}$,
where $Q^{\mathrm{co}}$ consists of the transposes of the squares of $Q$.
\end{rmk}

\begin{rmk}[mixed squares]
\label{rmk:biadjointable}
\gutentag{00A5}%
Let $\sigma$ be a commutative square whose horizontal edges are objects of $I$
and whose vertical edges are objects of $P$.
Read horizontally it can be a $1$-morphism of $I$,
and read vertically one of $P$; functors out of $\A\lrrefl{I}{P}$ then send it
to a left, resp.\ right, Beck--Chevalley square.
By \cref{lem:mates-conjugate} these conditions agree,
so it does not matter whether $\sigma$ is put into $I$, into $P$, or into both.
Putting $\sigma$ into $Q$ imposes more, as double Beck--Chevalley is a further
condition (\cref{rmk:double-BC}). This can also be done in two steps:
the mirror of $\gamma_{I}(\sigma)$ is a square of $\A\refl{I}$ whose vertical
edges are still objects of $\gamma_{I}(P)$,
and adding it to the second step imposes exactly that the double Beck--Chevalley
map of the image of $\sigma$ be invertible, since functors preserve mirrors.
So by \cref{prop:adjoin-univ},
$\A\lrqrefl{I}{P}{Q} \simeq \A\refl{I}\corefl{P'}$
for the locally full $P'$ generated by $\gamma_{I}(P)$ and the mirrors of the
$\gamma_{I}(\sigma)$, $\sigma \in Q$.
With $\Adj$ in place of $\Refl$ and $\Corefl$,
this is how Cnossen--Lenz--Linskens construct the universal biadjointable
functor \cite[Thm.~2.10]{cnossenlenzlinskens2025universal},
as we will explain in \cref{ex:plain-spans}.
\end{rmk}

\begin{exmp}
\label{ex:adjoin-basic}
\gutentag{0037}%
$(\Delta^{1})\corefl{e} \simeq \Corefl$
for the generic arrow $e$.
In general $\A\lrrefl{I}{P}$ is a free construction and not a
sub-$(\infty,2)$-category of $\A$. In \cref{subsec:2dloc-local} we will describe
the local objects inside a given $(\infty,2)$-category and compare them with
$\A\lrrefl{I}{P}$ (\cref{prop:local-vs-free});
unlike in \cref{prop:bousfield-pushout}(1), the two will differ in general,
even when the inclusion of the local objects has a left adjoint
(\cref{ex:local-vs-pushout}). \Cref{thm:2dloc-local} will identify the local
objects with the pushout in presentable $(\infty,2)$-categories. For small $\A$,
$I$ and $P$ this will give
$\Free(\A\lrrefl{I}{P}) \simeq \Pc_{y(I),y(P)} \subset \Free(\A)$
(\cref{rmk:free-local}).
\end{exmp}
